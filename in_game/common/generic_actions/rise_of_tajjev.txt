roj_declare_conquest_ambition = {
	type = situation
	potential = {
		scope:actor = {
			tag = A409
		}
	}
	allow = {
		scope:actor = {
			is_rebel_country = no
		}
	}

	automation_tick = never
	automation_tick_frequency = 12

	ai_tick = monthly
	ai_tick_frequency = 12

	cooldown = {
		type = roj_declare_conquest_ambition
		years = 5
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_tajjev = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = region
		target_flag = target
		name = "roj_declare_conquest_ambition_select_region"
		none_available_msg_key = "roj_declare_conquest_ambition_no_regions"
		column = {
			data = name
		}
		column = {
			data = population
		}
		visible = {
			scope:actor.capital.region != root
			any_area_in_region = {
				adjacent_to_owned_by = scope:actor
			}
		}
	}

	effect = {
		custom_tooltip = {
			text = roj_declare_conquest_ambition_tt
			scope:recipient = {
				if = {
					limit = { has_variable = roj_conquest_ambition }
					var:roj_conquest_ambition = {
						every_location_in_region = {
							remove_location_modifier = roj_tajjev_scare
						}
					}
				}
				set_variable = {
					name = roj_conquest_ambition
					value = scope:target
				}
			}
		}
	}

	ai_will_do = {
		if = {
			limit = {
				NOT = { scope:recipient = { has_variable = roj_conquest_ambition } }
				OR = {
					scope:target ?= region:amalyarat_region
					scope:target ?= region:iboros_region
				}
			}
			value = 1000
		}
		else = {
			value = -100
		}
	}
}

roj_plan_invasion = {
	type = situation
	potential = {
		scope:actor = {
			tag = A409
		}
	}
	allow = {
		scope:actor = {
			is_rebel_country = no
		}
	}

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_tajjev = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = country
		interaction_source_list = {
			if = {
				limit = {
					scope:recipient = {
						has_variable = roj_conquest_ambition
					}
				}
				scope:recipient = {
					var:roj_conquest_ambition = {
						every_country_with_capital_in_geography = {
							limit = { this != scope:actor }
							add_to_list = source
						}
					}
				}
			}
			scope:actor = {
				every_neighbor_country = {
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "roj_plan_invasion_select_country"
		none_available_msg_key = "roj_plan_invasion_no_countries"
		column = {
			data = name
		}
		column = {
			data = expected_army_size
		}
		visible = {
			is_real_country = yes
			scope:actor = {
				can_declare_war_on = root
				NOT = { has_truce_with = root }
				NOT = { is_allied_with = { target = root } }
			}
		}
	}

	cooldown = {
		type = roj_plan_invasion
		months = 3
	}

	automation_tick = never
	automation_tick_frequency = 12

	ai_tick = monthly
	ai_tick_frequency = {
		if = {
			limit = {
				at_war = yes
			}
			value = 18
		}
		else = {
			value = 3
		}
	}

	effect = {
		scope:actor = {
			if = {
				limit = {
					exists = scope:target
				}
				if = {
					limit = {
						scope:target = {
							capital = {
								OR = {
									region = scope:actor.capital.region
									AND = {
										scope:recipient = { has_variable = roj_conquest_ambition }
										region = scope:recipient.var:roj_conquest_ambition
									}
								}
							}
						}
					}
					drop_antagonism_bomb = {
						target = scope:target.capital
						modifier = antagonism_started_invasion
					}
					declare_war_with_cb = {
						target = scope:target
						type = casus_belli:cb_tajjev_conquests
					}
				}
				else = {
					add_casus_belli = {
						target = scope:target
						type = casus_belli:cb_tajjev_conquests
					}
					drop_antagonism_bomb = {
						target = scope:target.capital
						modifier = antagonism_planned_invasion
					}
				}
			}
		}
	}
	ai_will_do = {
		value = -20
		if = {
			limit = { exists = scope:target }
			if = {
				limit = {
					scope:target = {
						capital = {
							OR = {
								region = scope:actor.capital.region
								AND = {
									scope:recipient = { has_variable = roj_conquest_ambition }
									region = scope:recipient.var:roj_conquest_ambition
								}
							}
						}
					}
				}
				add = 200
			}
			add = {
				value = "scope:actor.opinion(scope:target)"
				multiply = -1
			}
			multiply = "scope:actor.relative_military_strength(scope:target)"
			if = {
				limit = {
					scope:target = { NOT = { is_neighbor_of = scope:actor } }
				}
				subtract = 1000
			}
		}
	}
}

roj_select_core_region = {
	type = situation
	potential = {
		scope:actor = {
			tag = A409
		}
	}
	price = price:rot_select_core_region_price
	allow = {
		scope:actor = {
			is_rebel_country = no
			custom_tooltip = {
				text = roj_has_NOT_selected_core_region_tt
				NOT = { has_variable = roj_has_selected_core_region }
			}
			at_war = no
		}
	}

	automation_tick = never
	automation_tick_frequency = 12

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_tajjev = this
			situation_is_active = yes
		}
	}

	select_trigger = {
		looking_for_a = region
		target_flag = target
		name = "roj_select_core_region_select_region"
		none_available_msg_key = "roj_select_core_region_no_regions"
		column = {
			data = name
		}
		column = {
			data = population
		}
		visible = {
			any_location_in_region = {
				owner ?= scope:actor
				percent >= 0.4
			}
		}
	}
	effect = {
		if = {
			limit = { exists = scope:target }
			custom_tooltip = {
				text = roj_select_core_region_tt_specific
				scope:actor = {
					set_variable = {
						name = roj_has_selected_core_region
						value = scope:target
					}
				}
				scope:target ?= {
					if = {
						limit = {
							scope:actor = {
								has_ruler = yes
								ruler ?= { has_character_modifier = tajjev_wants_to_play }
							}
						}
						every_location_in_region = {
							limit = { owner ?= scope:actor }
							change_integration_level = core
							add_location_modifier = {
								modifier = roj_tajjev_core_region
								years = -1
								mode = add_and_extend
							}
						}
					}
				}
			}
		}
	}

	ai_will_do = {
		add = {
			if = {
				limit = { exists = scope:target scope:target = region:amalyarat_region }
				value = 100
			}
		}
		add = {
			if = {
				limit = { exists = scope:target scope:target = region:iboros_region }
				value = 90
			}
		}
	}
}

roj_promote_regional_culture = {
	type = situation
	potential = {
		scope:actor = {
			tag = A409
		}
	}
	allow = {
		scope:actor = {
			is_rebel_country = no
			trigger_if = {
				limit = { has_variable = roj_has_selected_core_region }
				var:roj_has_selected_core_region = {
					any_ownable_location_in_region = {
						has_owner = yes
						owner = scope:actor
						count = all
					}
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = roj_has_selected_core_region_tt
					has_variable = roj_has_selected_core_region
				}
			}
			custom_tooltip = {
				text = roj_NOT_assumed_regional_culture_tt
				NOT = { has_variable = roj_assumed_regional_culture }
			}
		}
	}

	ai_tick = monthly
	ai_tick_frequency = 3

	automation_tick = never
	automation_tick_frequency = 12

	select_trigger = {
		looking_for_a = situation
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:rise_of_tajjev = this
			situation_is_active = yes
		}
	}

	effect = {
		scope:actor = {
			set_variable = roj_assumed_regional_culture
			change_culture = culture:takkev_culture
		}
	}

	ai_will_do = {
		add = 100
	}
}
