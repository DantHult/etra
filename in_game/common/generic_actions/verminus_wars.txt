verminus_press_claims = {
	type = situation
	show_message = no
	potential = {
	}

	allow = {
		scope:actor = {
			at_war = no
			has_raised_levies = no
			is_member_of_international_organization = international_organization:verminus_realms
		}
	}

	automation_tick = never
	automation_tick_frequency = 12
	ai_tick = monthly
	ai_tick_frequency = 3
	price = price:verminus_press_claims_price

	cooldown = {
		type = verminus_press_claims_cooldown_key
		years = 2
	}

	select_trigger = {
		looking_for_a = situation
		interaction_source_list = {
			situation:verminus_wars = {
				add_to_list = source
			}
		}
		target_flag = recipient
		name = "choose_situation"
		column = {
			data = name
		}
		visible = {
			situation:verminus_wars = this
			situation_is_active = yes
			trigger_if = {
				limit = { exists = scope:actor }
				scope:actor = {
					is_member_of_international_organization = international_organization:verminus_realms
				}
			}
		}
	}

	select_trigger = {
		looking_for_a = province
		target_flag = target
		name = "choose_province"
		none_available_msg_key = "no_provinces_available"
		column = { data = owner_flag }
		column = { data = name }
		column = { data = population }
		visible = {
			scope:actor = { can_declare_war_on = root.owner }
			owner = {
				is_member_of_international_organization = international_organization:verminus_realms
				NOT = { this = scope:actor }
				NOT = {
					OR = {
						overlord ?= scope:actor
						overlord ?= { is_subject_of = scope:actor }
						has_mutual_scripted_relation = {
							type = relation_type:alliance
							target = scope:actor
						}
					}
				}
			}
			any_location_in_province = {
				is_neighbor_of = scope:actor
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:target
			}
			custom_tooltip = {
				text = verminus_press_claims_explanation_tt
				scope:actor = {
					trigger_event_silently = verminus_wars.100
				}
			}
		}
		else_if = {
			limit = {
				NOT = { exists = scope:target }
			}
			custom_tooltip = verminus_press_claims_explanation_tt
		}
	}

	ai_will_do = {
		if = {
			limit = {
				exists = scope:target
			}

			if = {
				limit = {
					scope:actor = { has_casus_belli_on = scope:target.owner }
				}
				add = {
					value = -10000
					desc = "We already have a cb on them"
				}
			}

			if = {
				limit = {
					scope:actor = {
						has_truce_with = scope:target.owner
					}
				}
				add = {
					value = -1000
					desc = "truce"
				}
			}

			if = {
				limit = {
					scope:actor = { is_during_bankruptcy = yes }
				}
				subtract = {
					desc = "[bankruptcy|e]"
					value = 1000
				}
			}

			if = {
				limit = {
					scope:actor = { stability < 0 }
				}
				subtract = {
					desc = "low stability"
					value = 1000
				}
			}

			add = {
				desc = "RELATIVE_STRENGTH"
				value = "scope:actor.relative_defensive_alliance_strength(scope:target.owner)"
				multiply = 150
				subtract = 60
			}

			add = {
				desc = "we want their land"
				value = "scope:actor.conquer_desire(scope:target.owner)"
				divide = 5
			}

			if = {
				limit = {
					scope:actor = { NOT = { wants_to_attack = scope:target.owner } }
				}
				add = {
					value = -2000
					desc = "We dont want to attack them"
				}
			}
			else = {
				add = {
					value = 2000
					desc = "We want to attack them"
				}
			}
		}
		else = {
			add = {
				value = 25
				desc = BASE_VALUE
			}
		}
	}
}
