verminus_realms = {

	has_target = no
	unique = yes
	has_leader_country = yes
	disband_if_no_leader = no
	expel_members_who_are_targets_of_other_members = no
	antagonism_modifier_for_taking_land_from_fellow_member = 0.5
	leader_title_key = "VERMINUS_REALMS_LEADER"
	leader_type = character
	leader = {
		leader_country ?= {
			if = {
				limit = { has_regent = yes }
				heir = {
					add_to_list = leaders
				}
			}
			else = {
				ruler = {
					add_to_list = leaders
				}
			}
		}
	}
	leader_change_trigger_type = none
	leader_change_method = none
	gives_food_access_to_members = yes

	create_visible_trigger = {
		always = no
	}

	invite_visible_trigger = {
		always = no
	}

	can_lead_trigger = {
		culture = culture:rattolini_culture
	}

	can_declare_war = {
		trigger_if = {
			limit = {
				is_situation_active = situation:verminus_wars
			}
			always = yes
		}
		trigger_else = {
			NAND = {
				scope:attacker ?= {
					is_member_of_international_organization = international_organization:verminus_realms
				}
				scope:defender ?= {
					is_member_of_international_organization = international_organization:verminus_realms
				}
			}
		}
	}

	gives_military_access_to_all_when_at_war = yes

	can_join_trigger = {
		culture = culture:rattolini_culture
		capital ?= {
			area = area:freehold_2_area
		}
	}

	can_leave_trigger = {
		always = no
	}

	auto_leave_trigger = {
		always = no
	}

	auto_disband_trigger = {
		always = no
	}

	on_joined = {
	}

	on_left = {
	}

	monthly_effect = {
	}

	variables = {
	}

	ai_desire_to_join = {
	}

	ai_desire_to_allow_new_member = {
	}
}
