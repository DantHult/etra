on_papal_opinion_added = {
	trigger = {
		OR = {
			modifier:papal_relations < 0
			modifier:papal_relations > 0
		}
		country_exists = c:PAP
		not = { tag = PAP }
	}
	effect = {
		c:PAP = {
			remove_opinion = {
				target = prev
				modifier = opinion_papal_relations
			}
		}
		reverse_add_opinion = {
			target = c:PAP
			modifier = opinion_papal_relations
			scale = modifier:papal_relations
		}
	}
}

on_papal_opinion_removed = {
	trigger = {
		country_exists = c:PAP
		not = { tag = PAP }
		c:PAP = {
			has_opinion = {
				target = prev
				modifier = opinion_papal_relations
			}
		}
		modifier:papal_relations = 0
	}
	effect = {
		c:PAP = {
			remove_opinion = {
				target = prev
				modifier = opinion_papal_relations
			}
		}
	}
}

on_papal_authority_updated = {
	trigger = {
		tag = PAP
		country_exists = c:PAP
		is_member_of_international_organization = international_organization:catholic_church
	}
	effect = {
		if = {
			limit = {
				international_organization:catholic_church = {
					var:papal_authority >= 75
				}
			}
			religion:catholic = {
				add_religion_modifier = {
					modifier = high_papal_authority
					days = -1
				}
				remove_religion_modifier = low_papal_authority
			}
		}
		else_if = {
			limit = {
				international_organization:catholic_church = {
					var:papal_authority < 25
				}
			}
			religion:catholic = {
				add_religion_modifier = {
					modifier = low_papal_authority
					days = -1
				}
				remove_religion_modifier = high_papal_authority
			}
		}
		else = {
			religion:catholic = {
				remove_religion_modifier = high_papal_authority
				remove_religion_modifier = low_papal_authority
			}
		}
	}
}

}

			}

				has_country_modifier = fra_tight_grip_over_neapolitan_court

		}			
		if = {
			limit = {
				has_country_modifier = fra_neapolitan_court_autonomy
			}
			remove_country_modifier = fra_neapolitan_court_autonomy
		}	
		c:NAP = {
			if = {
				limit = {
					has_country_modifier = nap_aligned_neapolitan_court
				}
				remove_country_modifier = nap_aligned_neapolitan_court
			}	
			if = {
				limit = {
					has_country_modifier = nap_tight_grip_over_neapolitan_court
				}

			}

				limit = {
					has_country_modifier = nap_neapolitan_court_autonomy
				}

on_fraticide_underage_succession_waiting = {
	trigger = {
		has_variable = fraticide_succession_has_child_sons
	}
	effect = {
		if = {
			limit = {
				list_size = { name = rival_male_children value > 0 }
			}
			if = {
				limit = {
					any_in_list = {
						list = rival_male_children
						is_alive = yes
						is_adult = yes
					}
				}
				trigger_event_silently = succession.5
			}
			else = {
				every_in_list = {
					list = rival_male_children
					limit = { is_alive = no }
					remove_from_list = rival_male_children
				}
			}
		}
		else = {
			trigger_event_silently = succession.5
		}
	}
}

# A410 only. The pulse is the monthly tick: 1 March (month 3) and 1 September (month 9).
# current_month runs 1 through 12. Swap only when the sitting ruler is the wrong line.
issarsiya_diarchy_monthly = {
	trigger = {
		tag = A410
		has_ruler = yes
		ruler = { has_variable = issarsiya_diarch_line }
		OR = {
			AND = {
				current_month = 3
				ruler = { var:issarsiya_diarch_line = 1 }
				OR = {
					AND = {
						has_variable = issarsiya_fire_diarch
						var:issarsiya_fire_diarch = { is_alive = yes }
					}
					AND = {
						has_heir = yes
						heir = {
							is_alive = yes
							has_variable = issarsiya_diarch_line
							var:issarsiya_diarch_line = 2
						}
					}
				}
			}
			AND = {
				current_month = 9
				ruler = { var:issarsiya_diarch_line = 2 }
				OR = {
					AND = {
						has_variable = issarsiya_frost_diarch
						var:issarsiya_frost_diarch = { is_alive = yes }
					}
					AND = {
						has_heir = yes
						heir = {
							is_alive = yes
							has_variable = issarsiya_diarch_line
							var:issarsiya_diarch_line = 1
						}
					}
				}
			}
		}
	}
	effect = {
		if = {
			limit = { ruler = { var:issarsiya_diarch_line = 1 } }
			if = {
				limit = { NOT = { has_variable = issarsiya_frost_diarch } }
				set_variable = { name = issarsiya_frost_diarch value = ruler }
			}
			else_if = {
				limit = { NOT = { var:issarsiya_frost_diarch = ruler } }
				set_variable = { name = issarsiya_frost_diarch value = ruler }
			}
		}
		if = {
			limit = { ruler = { var:issarsiya_diarch_line = 2 } }
			if = {
				limit = { NOT = { has_variable = issarsiya_fire_diarch } }
				set_variable = { name = issarsiya_fire_diarch value = ruler }
			}
			else_if = {
				limit = { NOT = { var:issarsiya_fire_diarch = ruler } }
				set_variable = { name = issarsiya_fire_diarch value = ruler }
			}
		}
		if = {
			limit = {
				has_heir = yes
				heir = {
					is_alive = yes
					has_variable = issarsiya_diarch_line
					var:issarsiya_diarch_line = 2
				}
			}
			if = {
				limit = { NOT = { has_variable = issarsiya_fire_diarch } }
				set_variable = { name = issarsiya_fire_diarch value = heir }
			}
			else_if = {
				limit = { var:issarsiya_fire_diarch = { is_alive = no } }
				set_variable = { name = issarsiya_fire_diarch value = heir }
			}
		}
		if = {
			limit = {
				has_heir = yes
				heir = {
					is_alive = yes
					has_variable = issarsiya_diarch_line
					var:issarsiya_diarch_line = 1
				}
			}
			if = {
				limit = { NOT = { has_variable = issarsiya_frost_diarch } }
				set_variable = { name = issarsiya_frost_diarch value = heir }
			}
			else_if = {
				limit = { var:issarsiya_frost_diarch = { is_alive = no } }
				set_variable = { name = issarsiya_frost_diarch value = heir }
			}
		}
		ruler = { save_scope_as = issarsiya_outgoing }
		if = {
			limit = { current_month = 3 }
			var:issarsiya_fire_diarch = { save_scope_as = issarsiya_incoming }
		}
		else = {
			var:issarsiya_frost_diarch = { save_scope_as = issarsiya_incoming }
		}
		issarsiya_reseat_lines_effect = yes
		trigger_event_non_silently = issarsiya_diarchy.1
	}
}
