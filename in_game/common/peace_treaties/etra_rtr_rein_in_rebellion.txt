etra_rtr_rein_in_rebellion = {
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 80
		}
	}

	potential = {
		scope:war = {
			is_no_cb_war = no
			OR = {
				casus_belli ?= casus_belli:cb_etra_red_turban_rebellions_rein_in_rebels
				casus_belli ?= casus_belli:cb_etra_red_turban_rebellions
			}
		}
		scope:winner = {
			tag = A378
		}
		scope:loser = {
			modifier:can_create_red_turban_rebellions_cb = yes
		}
	}
	allow = {
	}

	effect = {
		scope:loser = {
			etra_rtr_change_allegiance_points = {
				value = 200
			}
			if = {
				limit = { is_member_of_international_organization = international_organization:etra_red_turban_rebels }
				international_organization:etra_red_turban_rebels = {
					remove_country_from_international_organization = scope:loser
				}
			}
			custom_tooltip = etra_rtr_rein_in_rebellion_tt
		}
	}

	ai_desire = {
		value = 1000
	}
}

etra_rtr_demand_occupied_territories = {
	cost = {
		add = {
			desc = "DIPLOREASON_BASE"
			value = 80
		}
		add = {
			desc = "DIPLOREASON_OCCUPATION_VALUE"
			value = {
				value = 0
				scope:loser = {
					every_owned_location = {
						limit = { controller = scope:winner }
						add = location_tax_base
						divide = 5
					}
				}
			}
		}
	}

	base_antagonism = {
		value = {
			value = 1
			scope:loser = {
				every_owned_location = {
					limit = { controller = scope:winner }
					add = location_tax_base
					divide = 10
				}
			}
		}
	}
	antagonism_type = antagonism_peace_rtr_demand_all_occupied_territories
	potential = {
		scope:war = {
			is_no_cb_war = no
			OR = {
				casus_belli ?= casus_belli:cb_etra_red_turban_rebellions_rein_in_rebels
				casus_belli ?= casus_belli:cb_etra_red_turban_rebellions
			}
		}
		scope:winner = {
			country_type = location
			modifier:can_create_red_turban_rebellions_cb = yes
		}
		scope:loser = {
			country_type = location
			tag = A378
		}
	}
	allow = {
		scope:winner = {
			num_locations >= 25
		}
		scope:loser = {
			any_owned_location = {
				controller = scope:winner
				count >= 25
			}
		}
	}

	effect = {
		if = {
			limit = {
				exists = scope:winner
				exists = scope:loser
			}
			scope:winner = {
				custom_tooltip = {
					text = etra_rtr_demand_occupied_territories_tt
					every_controlled_location = {
						limit = {
							owner = scope:loser
						}
						change_location_owner = scope:winner
						# etra: flavor_chi.32 dropped (YUA/CHI capital-capture event)
					}
				}
			}
		}
		else = { custom_tooltip = etra_rtr_demand_occupied_territories_no_winner_loser_tt }
	}

	ai_desire = {
		value = 1000
	}
}
