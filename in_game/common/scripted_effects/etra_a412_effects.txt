a412_add_notoriety_effect = {
	if = {
		limit = { NOT = { has_variable = a412_notoriety } }
		set_variable = { name = a412_notoriety value = 0 }
	}
	change_variable = {
		name = a412_notoriety
		add = $VALUE$
	}
	if = {
		limit = {
			var:a412_notoriety >= 40
			NOT = { has_variable = a412_notoriety_threshold }
			has_variable = a412_ritual_i_done
		}
		set_variable = { name = a412_notoriety_threshold value = 1 }
		trigger_event_non_silently = flavor_A412.25
	}
}

	set_variable = { name = a412_ritual_i_done value = 1 }
