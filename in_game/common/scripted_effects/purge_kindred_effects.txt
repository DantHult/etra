purge_kindred_monthly_effect = {

	every_cabinet = {

		limit = {

			has_cabinet_action = yes

			cabinet_action = cabinet_action:purge_kindred

		}

		interaction_target:target = {

			save_temporary_scope_as = purge_kindred_group

		}

		interaction_target:target_1 = {

			every_location_in_province = {

				every_pop = {

					limit = {

						culture = {

							has_culture_group = scope:purge_kindred_group

							NOT = { is_accepted_in = owner }

						}

					}

					add_pop_size = {

						value = pop_size

						multiply = -0.02

					}

				}

			}

		}

	}

	purge_kindred_monthly_satisfaction_effect = yes

}

	}

}
