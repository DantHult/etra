verminus_wars_init_timers = {
	if = {
		limit = {
			NOT = { has_variable = verminus_wars_timers_initialized }
		}
		set_variable = verminus_wars_timers_initialized
		set_variable = { name = verminus_wars_minimum_duration_timer value = 1 years = 10 }
		set_variable = { name = verminus_wars_max_duration_timer value = 1 years = 25 }
	}
}

verminus_wars_unify_winner = {
	situation:verminus_wars = {
		set_variable = {
			name = verminus_wars_winner
			value = this
		}
	}
	form_country = formable_country:Rattopaese_f
}

verminus_wars_save_losers_list = {
	clear_global_variable_list = verminus_wars_losers_list
	international_organization:verminus_realms ?= {
		every_international_organization_member = {
			limit = {
				this != situation:verminus_wars.var:verminus_wars_winner
			}
			add_to_global_variable_list = {
				name = verminus_wars_losers_list
				target = this
			}
		}
	}
}
