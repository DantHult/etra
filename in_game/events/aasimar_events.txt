namespace = aasimar_events

aasimar_events.1 = {
	type = country_event
	title = aasimar_events.1.title
	desc = aasimar_events.1.desc
	trigger = {
		NOT = { culture = { has_culture_group = culture_group:aasimar_group } }
		any_owned_location = {
			has_aasimar = yes
			is_capital = no
		}
	}
	illustration_tags = { 10 = regular  10 = interior }
	immediate = {
		event_illustration_government_estate_effect = yes
		random_owned_location = {
			limit = { has_aasimar = yes  is_capital = no }
			save_scope_as = racial_pop_location_target
		}
	}
	option = { name = aasimar_events.1.a  ai_chance = { factor = 50 }
		add_government_power = government_power_severe_penalty
		change_societal_value = { type = xenophile_vs_xenophobe value = societal_value_minor_move_to_left }
		if = { limit = { exists = scope:racial_pop_location_target }
			scope:racial_pop_location_target = { change_control = control_severe_penalty }
		}
	}
	option = { name = aasimar_events.1.b  ai_chance = { factor = 50 }
		add_government_power = government_power_mild_penalty
		if = { limit = { exists = scope:racial_pop_location_target }
			scope:racial_pop_location_target = { change_control = control_mild_penalty }
		}
	}
	option = { name = aasimar_events.1.c  ai_chance = { factor = 50 }
		change_societal_value = { type = xenophile_vs_xenophobe value = societal_value_minor_move_to_right }
	}
	option = { name = aasimar_events.1.e
		trigger = { ruler_or_regent = { has_personality = charismatic_negotiator_personality } }
		highlight = yes  ai_chance = { factor = 25 }
		add_government_power = government_power_weak_penalty
		change_societal_value = { type = xenophile_vs_xenophobe value = societal_value_minor_move_to_left }
		if = { limit = { exists = scope:racial_pop_location_target }
			scope:racial_pop_location_target = { change_control = control_weak_penalty }
		}
	}
}
