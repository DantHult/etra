# 1. The empty tomb
flavor_A325.1 = {
	hide_portraits = yes
	type = country_event
	title = flavor_A325.1.title
	desc = flavor_A325.1.desc
	fire_only_once = yes
	outcome = neutral
	dynamic_historical_event = {
		tag = A325
		from = 1338.1.1
		to = 9999.1.1
		monthly_chance = 12
	}

	trigger = {
		tag = A325
		owns = location:etra304
		at_war = no
		is_during_bankruptcy = no
		exists = c:A331
	}

	illustration_tags = {
		10 = regular
		10 = interior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:clergy_estate }
		location:etra66 = { save_scope_as = target_location }
		c:A331 = { save_scope_as = target_country }
		ruler_or_regent = { save_scope_as = target_character }
	}

	option = {
		name = flavor_A325.1.a
		historical_option = yes
		add_legitimacy = legitimacy_mild_bonus
		add_prestige = prestige_weak_penalty
	}

	option = {
		name = flavor_A325.1.b
		add_prestige = prestige_weak_bonus
	}
}
