namespace = flavor_a412

# Opening of the Drowned Fleet. EU4 war_unique_intro_events.37 handed navy tradition,
# mercantilism, and a claim on every coastal center of trade in the world.
# Cores are the EU5 conquest claim, and they stop at Typhrea.

flavor_a412.1 = {
	type = country_event
	title = flavor_a412.1.title
	desc = flavor_a412.1.desc
	fire_only_once = yes
	outcome = neutral

	trigger = {
		tag = A412
	}

	illustration_tags = {
		10 = armed
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:burghers_estate }
	}

	option = {
		name = flavor_a412.1.a
		historical_option = yes
		ai_chance = { value = 100 }
		add_navy_tradition = navy_tradition_extreme_bonus
		change_societal_value = {
			type = mercantilism_vs_free_trade
			value = societal_value_minor_move_to_left
		}
		custom_tooltip = flavor_a412.1.a.tt
		hidden_effect = {
			region:typhrea_region = {
				every_location_in_region = {
					limit = {
						is_coastal = yes
						is_market_center = yes
						exists = owner
						NOT = { owner = root }
					}
					add_core = root
				}
			}
		}
	}
}

# A thing from the deep at a harbor. vampire_events.103

	outcome = neutral

	trigger = {
		tag = A412

		any_owned_location = { is_coastal = yes }
		NOT = { has_variable = a412_kraken_cooldown }

	}

		event_illustration_government_estate_effect = yes

		set_variable = { name = a412_kraken_cooldown value = yes days = 1825 }

	option = {
		name = flavor_a412.103.b

			change_prosperity = prosperity_severe_penalty
		}
		set_variable = { name = a412_kraken_cooldown value = yes days = 1825 }
	}
}

# Spend on the harbor, or on the merchants. vampire_events.104
# EU5 markets are not tiered centers of trade; a dock is the harbor upgrade.
flavor_a412.104 = {
	type = country_event
	title = flavor_a412.104.title
	desc = flavor_a412.104.desc
	fire_only_once = no
	outcome = neutral

	trigger = {
		tag = A412
		has_reform = government_reform:dread_fleet_reform
		any_owned_location = {
			is_coastal = yes
			NOT = { has_building = building_type:dock }
		}
		NOT = { has_variable = a412_dock_cooldown }
	}

	illustration_tags = {
		10 = regular
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:burghers_estate background = estate_type:nobles_estate }

			limit = {
				is_coastal = yes
				NOT = { has_building = building_type:dock }
			}

		}
	}

	option = {
		name = flavor_a412.104.a

			construct_building = {
