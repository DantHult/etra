namespace = etra_hre

# Hidden one-time scheduler on the Commonweal leader; broadcasts to human members after 1–5 years.
etra_hre.0 = {
	type = country_event
	hidden = yes
	title = etra_hre.0.title

	trigger = {
		exists = international_organization:hre
		is_leader_of_international_organization = international_organization:hre
		has_global_variable = etra_commonweal_pulse_scheduled
		NOT = { has_global_variable = etra_commonweal_pulse_fired }
	}

	immediate = {
		etra_broadcast_commonweal_pulse_to_human_members = yes
		set_global_variable = etra_commonweal_pulse_fired
		remove_global_variable = etra_commonweal_pulse_scheduled
	}

	option = {
		name = etra_hre.0.a
		ai_chance = { factor = 100 }
	}
}
