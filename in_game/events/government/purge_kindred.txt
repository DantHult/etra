namespace = purge_kindred

purge_kindred.10 = {
	type = country_event
	title = purge_kindred.10.title
	desc = purge_kindred.10.desc
	outcome = negative

	trigger = {
		has_active_purge_kindred_cabinet = yes
		NOT = { has_variable = purge_kindred_migration_cooldown }
	}

	illustration_tags = {
		10 = angry
		10 = exterior
	}

	immediate = {
		event_illustration_estate_effect = {
			foreground = estate_type:peasants_estate
			background = estate_type:peasants_estate
		}
		purge_kindred_save_active_scopes = yes
		set_variable = { name = purge_kindred_migration_cooldown days = 365 }
	}

	option = {
		name = purge_kindred.10.a
		ai_chance = { factor = 40 }
		if = {
			limit = {
				exists = scope:target_province
			}
			scope:target_province = {
				every_location_in_province = {
					add_migration = {
						owner = root
						from = this.province.province_definition
						to = root.capital.province_definition
						amount = 0.3
						months = 6
					}
				}
			}
		}
		change_societal_value = {
			type = xenophile_vs_xenophobe
			value = societal_value_minor_move_to_right
		}
	}

	option = {
		name = purge_kindred.10.b
		ai_chance = { factor = 60 }
		if = {
			limit = {
				exists = scope:target_province
			}
			scope:target_province = {
				every_location_in_province = {
					change_control = control_mild_penalty
				}
			}
		}
		purge_kindred_monthly_satisfaction_effect = yes
	}
}

				NOT = { this = root }

		name = purge_kindred.13.a

			reverse_add_opinion = {
				target = scope:condemning_country
				modifier = opinion_negative_societal_values
			}
		}
	}

	option = {
		name = purge_kindred.13.b
		ai_chance = { factor = 30 }
