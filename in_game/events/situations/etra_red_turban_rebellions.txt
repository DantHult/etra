scripted_effect etra_rtr_released_country_setup_effects = {
	set_variable = { name = etra_rtr_iborros_allegiance value = 0 }
	change_government_type = government_type:monarchy
	add_reform = government_reform:etra_warring_state
	add_to_global_variable_list = {
		name = etra_rtr_faction_list
		target = this
	}

	change_heir_selection = heir_selection:cognatic_primogeniture
	add_estate_satisfaction = { type = estate_type:nobles_estate value = estate_satisfaction_radical_bonus }
	add_estate_satisfaction = { type = estate_type:clergy_estate value = estate_satisfaction_radical_bonus }
	add_estate_satisfaction = { type = estate_type:peasants_estate value = estate_satisfaction_radical_bonus }
	add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_radical_bonus }
	# etra: confucian/daoist sect IO join dropped for manyfold
	if = {
		limit = { exists = scope:target_location }
		set_capital = scope:target_location
	}
	save_temporary_scope_as = temp_warlord_country
	if = {
		limit = { exists = scope:target_character }
		scope:target_character = {
			move_country = scope:temp_warlord_country
			every_child = {
				limit = { NOT = { owner = scope:temp_warlord_country } }
				move_country = scope:temp_warlord_country
			}
		}
		set_new_ruler = scope:target_character
	}
	every_province = {
		limit = { any_location_in_province = { NOT = { integration_level = core } } }
		change_province_integration = core
		change_province_food_percentage = 1
	}
	trigger_event_silently = {
		id = etra_red_turban_rebellions.22
		days = 35	#It must trigger after the first monthly tick
	}
}

scripted_effect etra_rtr_release_warlord_tag = {
	if = {
		limit = { NOT = { country_exists = c:$tag$ } }
		custom_tooltip = {
			text = etra_rtr_release_warlord_country_tt
			remove_variable = etra_rtr_release_tag_$tag$
			change_variable = {
				name = etra_rtr_historical_countries_to_release
				subtract = 1
			}
			create_country_from_cores_in_our_locations = c:$tag$
			c:$tag$ = {
				save_scope_as = target_country
				change_country_color = map_$tag$
				change_culture = $culture$
				etra_rtr_released_country_setup_effects = yes
				add_accepted_culture = $accepted_culture$
				# etra: china_chinese_unification area preference dropped
			}
		}
	}
	custom_description = {
		text = etra_rtr_lose_competent_characters
		value = $culture$
		object = c:$tag$
		ordered_character = {
			order_by = total_abilities
			limit = {
				culture = $culture$
				is_ruler = no
				is_heir = no
				is_loyal = yes
				in_cabinet = no
				is_general = no
				is_explorer = no
				is_admiral = no
				is_consort = no
			}
			max = 8
			check_range_bounds = no
			move_country = c:$tag$
		}
	}
}

scripted_effect etra_rtr_prepare_cultural_warlord_tag = {
	culture:$culture$ = { save_scope_as = target_culture }
	remove_variable = etra_rtr_release_$culture$
	ordered_owned_location = {
		limit = {
			NOT = { dominant_culture = root.culture }
			dominant_culture = scope:target_culture
			province != root.capital.province
		}
		order_by = {
			value = population
			if = {
				limit = { location_rank = location_rank:town }
				multiply = 5
			}
			if = {
				limit = { OR = { location_rank = location_rank:city  location_rank = location_rank:megalopolis } }
				multiply = 10
			}
		}
		check_range_bounds = no
		max = 1
		save_scope_as = target_location
		create_dynasty_from_location = random
		last_dynasty_in_location = { save_scope_as = target_dynasty }
		create_country_from_location = {
			save_scope_as = target_country
			etra_rtr_released_country_setup_effects = yes
			set_capital = prev
			create_character = {
				culture = scope:target_culture
				age = 25
				dynasty = scope:target_dynasty
				save_scope_as = target_character
				prev = { set_new_ruler = prev }
			}
		}
	}
}
scripted_effect etra_rtr_release_cultural_warlord_tag = {
	if = {
		limit = {
			exists = scope:target_culture
			exists = scope:target_location
			exists = scope:target_country
		}
		custom_tooltip = {
			text = etra_rtr_release_cultural_warlord_tag_tt
			every_owned_location = {
				limit = {
					NOT = { dominant_culture = root.culture }
					dominant_culture = scope:target_culture
					province != root.capital.province
					NOT = {	#Checks that the location is not owned by a pop based country already
						any_nomad_countries_in_location = {
							count > 0
						}
					}
				}
				change_location_owner = scope:target_country
				add_core = scope:target_country
			}
			#clean up disconnected locations
			#It is done like this because is_connected_to = scope:target_location works not due to the location being no longer part of Yuán once the tag is released
			scope:target_country = {
				every_owned_location = {
					limit = { NOT = { is_connected_to = scope:target_location } }
					if = {
						limit = { is_core_of = scope:target_country }
						remove_core = scope:target_country
					}
					change_location_owner = root
					change_control = control_radical_bonus
				}
				change_culture = scope:target_culture
			}
		}
	}
}

scripted_effect etra_rtr_create_io = {
	if = {
		limit = {
			NOT = { exists = international_organization:etra_red_turban_rebels }
		}
		create_international_organization = {
			type = international_organization_type:etra_red_turban_rebels
			hidden_effect = {
				add_country_to_international_organization = prev
				set_leader_country = prev
				set_variable = etra_rtr_empty_rebellion
			}
		}
	}
}

scripted_effect etra_rtr_add_country_to_rebels_and_dow_iborros = {
	add_stability = stability_severe_penalty
	$country$ = {
		etra_rtr_create_io = yes
		if = {
			limit = { exists = international_organization:etra_red_turban_rebels }
			if = {
				limit = { international_organization:etra_red_turban_rebels = { has_variable = etra_rtr_empty_rebellion } }
				international_organization:etra_red_turban_rebels = {
					remove_variable = etra_rtr_empty_rebellion
					set_leader_country = prev
				}
			}
			else = {
				international_organization:etra_red_turban_rebels = {
					add_country_to_international_organization_no_update = prev
					if = {
						limit = { international_organization_has_leader = no }
						set_leader_country = prev
					}
				}
			}
			hidden_effect = {
				international_organization:etra_red_turban_rebels = {
					every_international_organization_member = {
						add_truce_with = { target = $country$ mutual = yes months = 6 }
					}
				}
			}
		}
		add_opinion = {
			modifier = opinion_etra_rtr_red_turban_rebellions
			target = c:A378
		}
	}
	if = {
		limit = {
			can_declare_legal_war_on = { target = $country$ }
		}
		declare_war_with_cb = {
			target = $country$
			type = casus_belli:cb_etra_red_turban_rebellions_rein_in_rebels
		}
	}
	else = {
		add_casus_belli = {
			target = $country$
			type = casus_belli:cb_etra_red_turban_rebellions_rein_in_rebels
		}
	}
	$country$ = {
		raise_all_levies = { type = army instant = yes }
	}
}

scripted_trigger etra_rtr_can_release_tag = {
	var:etra_rtr_historical_countries_to_release > 0
	has_variable = etra_rtr_release_tag_$tag$
	NOT = { country_exists = c:$tag$ }
	owns = location:$location$
	location:$location$ = { is_capital = no }
}

scripted_trigger etra_rtr_is_loyal = {
	has_variable = etra_rtr_iborros_allegiance
	var:etra_rtr_iborros_allegiance >= ETRA_RTR_IS_LOYAL_VALUE
}

scripted_trigger etra_rtr_is_neutral = {
	has_variable = etra_rtr_iborros_allegiance
	var:etra_rtr_iborros_allegiance >= ETRA_RTR_IS_NEUTRAL_VALUE
	var:etra_rtr_iborros_allegiance < ETRA_RTR_IS_LOYAL_VALUE
}

scripted_trigger etra_rtr_is_rebellious = {
	has_variable = etra_rtr_iborros_allegiance
	var:etra_rtr_iborros_allegiance < ETRA_RTR_IS_NEUTRAL_VALUE
}

#The Red Turban Rebellions (China event)
etra_red_turban_rebellions.1 = {
	type = country_event
	category = situation_event
	title = etra_red_turban_rebellions.1.title
	desc = etra_red_turban_rebellions.1.desc
	outcome = neutral
	image = "gfx/interface/illustrations/situation/red_turban_rebellions.dds"
	immediate = {
		save_scope_as = target_country
		ruler = { save_scope_as = target_ruler }
		if = {
			limit = { current_year < 1365 }
			create_character = {
				first_name = name_ying2.name_yu4
				last_name = name_peng
				estate = estate_type:clergy_estate
				culture = culture:selorian_culture
				religion = religion:manyfold
				birth_date = 1310 #Unknown but starts revolts 1338
				adm = {50 70}
				dip = {50 70}
				mil = {50 70}
				script = etra_rtr_rebel_monk
				add_character_modifier = { modifier = rebel_leader_modifier years = -1 mode = add_and_extend }
				save_scope_as = target_character
			}
		}
		else = {
			create_character = {
				estate = estate_type:clergy_estate
				culture = culture:selorian_culture
				religion = religion:manyfold
				age = 35
				adm = {50 70}
				dip = {50 70}
				mil = {50 70}
				script = etra_rtr_rebel_monk
				add_character_modifier = { modifier = rebel_leader_modifier years = -1 mode = add_and_extend }
				save_scope_as = target_character
			}
		}
		etra_rtr_setup_countries_to_release = yes
	}
	option = {
		name = etra_red_turban_rebellions.1.a
		custom_tooltip = etra_red_turban_rebellions.1.a.tt
		add_stability = stability_extreme_penalty
		add_legitimacy = legitimacy_extreme_penalty
		change_iboric_authority = { value = -20 }
		add_country_modifier = {
			modifier = etra_rtr_red_turban_rebellions
			years = -1
			mode = add_and_extend
			size = {
				if = {
					limit = {
						has_variable = etra_rtr_modifier_intensity
						var:etra_rtr_modifier_intensity > 0
					}
					value = var:etra_rtr_modifier_intensity
				}
				else = {
					value = 1
				}
			}
			desc = etra_rtr_red_turban_rebellion_duration
		}
		custom_tooltip = etra_red_turban_rebellions.1.a.at
	}
}
