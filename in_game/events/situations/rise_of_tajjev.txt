rise_of_tajjev.3 = {
	type = country_event
	category = situation_event

	desc = rise_of_tajjev.3.desc
	outcome = neutral
	illustration_tags = {
		10 = happy
		10 = exterior
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:peasants_estate background = estate_type:peasants_estate }
		c:A409 = {
			save_scope_as = tajjev_country
			capital ?= {
				save_scope_as = target_location
			}
		}
		global_var:tajjev_character = {
			save_scope_as = target_character
		}
		save_scope_as = target_country
	}
	option = {
		name = rise_of_tajjev.3.a
		disable_situation_effect = { type = rise_of_tajjev }
		add_stability = stability_severe_bonus
		add_prestige = prestige_severe_bonus
		if = {
			limit = { has_global_variable = tajjev_character global_var:tajjev_character = { is_alive = yes } }
			kill_character = {
				target = scope:target_character
				reason = execution
			}
		}
	}
}

rise_of_tajjev.100 = {
	type = country_event
	category = situation_event
	title = rise_of_tajjev.100.title
	desc = rise_of_tajjev.100.desc
	outcome = neutral
	illustration_tags = {
		10 = armed
		10 = exterior
	}
	immediate = {
		event_illustration_estate_effect = { foreground = estate_type:nobles_estate background = estate_type:nobles_estate }
		random_neighbor_country = {
			limit = {
				NOT = { has_truce_with = root }
				NOT = { has_target_casus_belli_on_us = root }
				NOT = { is_allied_with = { target = root } }
				NOT = { is_subject_of = root }
				is_rebel_country = no
				can_declare_war_on = root
				military_strength < root.military_strength
			}
			save_scope_as = target_country
		}
	}
	option = {
		name = rise_of_tajjev.100.a
		trigger = { exists = scope:target_country }
		declare_war_with_cb = {
			target = scope:target_country
			type = casus_belli:cb_tajjev_conquests
		}
	}
	option = {
		name = rise_of_tajjev.100.b
		add_prestige = prestige_weak_penalty
	}
}
