verminus_wars.2 = {
	type = country_event
	category = situation_event
	title = verminus_wars.2.title
	desc = verminus_wars.2.desc.winner
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/byzantine_succession_crisis.dds"

	immediate = {
		save_scope_as = from_country
		ruler_or_regent ?= { save_scope_as = target_character }
	}

	option = {
		name = verminus_wars.2.a
		trigger = {
			any_in_global_list = {
				variable = verminus_wars_losers_list
				country_exists = yes
				is_subject = no
			}
		}
		custom_tooltip = verminus_wars.2.a.tt
		hidden_effect = {
			every_in_global_list = {
				variable = verminus_wars_losers_list
				limit = {
					country_exists = yes
					is_subject = no
				}
				trigger_event_non_silently = verminus_wars.22
			}
		}
		add_prestige = prestige_mild_bonus
		add_government_power = government_power_mild_bonus

		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				num_subjects >= 2
			}
			modifier = {
				factor = 0.5
				army_size > 50
			}
		}
	}

	option = {
		name = verminus_wars.2.b
		trigger = {
			any_in_global_list = {
				variable = verminus_wars_losers_list
				country_exists = yes
				is_subject = no
			}
		}
		custom_tooltip = verminus_wars.2.b.tt
		hidden_effect = {
			every_in_global_list = {
				variable = verminus_wars_losers_list
				limit = {
					country_exists = yes
					is_subject = no
				}
				scope:from_country = {
					add_casus_belli = {
						target = prev
						type = casus_belli:cb_verminus_subjugate
					}
				}
			}
		}
		add_prestige = prestige_mild_bonus

		ai_chance = {
			factor = 50
			modifier = {
				factor = 2
				army_size > 50
			}
		}
	}

	option = {
		name = verminus_wars.2.c
		trigger = {
			NOT = {
				any_in_global_list = {
					variable = verminus_wars_losers_list
					country_exists = yes
					is_subject = no
				}
			}
		}
		add_prestige = prestige_severe_bonus
		add_government_power = government_power_mild_bonus
		add_legitimacy = legitimacy_mild_bonus
	}
}

verminus_wars.4 = {
	type = country_event
	category = situation_event
	title = verminus_wars.4.title
	desc = verminus_wars.4.desc
	outcome = neutral
	image = "gfx/interface/illustrations/disaster/byzantine_succession_crisis.dds"

	trigger = {
		is_situation_active = situation:verminus_wars
		international_organization:verminus_realms = {
			any_international_organization_member = {
				this != root
				NOT = { is_at_war_with = root }
				NOT = { has_truce_with = root }
				NOT = {
					root = { has_casus_belli_on = prev }
				}
			}
		}
	}

	immediate = {
		international_organization:verminus_realms = {
			random_international_organization_member = {
				limit = {
					this != root
					NOT = { is_at_war_with = root }
					NOT = { has_truce_with = root }
					NOT = {
						root = { has_casus_belli_on = prev }
					}
				}
				save_scope_as = target_enemy
			}
		}
	}

	option = {
		name = verminus_wars.4.a
		add_casus_belli = { target = scope:target_enemy type = casus_belli:cb_verminus_succession }
	}
}
